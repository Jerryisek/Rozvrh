                        </select>

                    </div>

                    <div class="grid grid-cols-2 gap-2">

                        <div>

                            <label class="block text-xs font-bold text-gray-700 mb-1">Typ</label>

                            <select v-model="newForm.type" class="w-full border rounded p-1 text-sm" required>

                                <option value="Přednáška">Přednáška (P)</option>

                                <option value="Cvičení">Cvičení (C)</option>

                                <option value="Laboratoř">Laboratoř (L)</option>

                            </select>

                        </div>

                        <div>

                            <label class="block text-xs font-bold text-gray-700 mb-1">Místnost</label>

                            <input type="text" v-model="newForm.room" placeholder="např. POREB130" class="w-full border rounded p-1 text-sm" required>

                        </div>

                    </div>

                    <div>

                        <label class="block text-xs font-bold text-gray-700 mb-1">Den</label>

                    <div class="grid grid-cols-2 gap-2">

                                <option v-for="(time, index) in timeBlocks" :value="index">{{ time }}</option>

                            </select>

                        </div>

                            {{ time }}

                        </th>

                    </tr>

                </thead>

                <tbody>

                    <tr v-for="day in days" :key="day">

                        <td class="border bg-gray-100 font-bold p-2 uppercase text-xs">{{ day }}</td>

                       

                        <template v-for="(cell, index) in computedGrid[day]" :key="index">

                            <td v-if="cell === null" class="border border-gray-200 p-2"></td>

                           

                            <td v-else-if="cell !== 'skip'"

                                :colspan="cell.span"

                                class="border border-gray-300 p-2 relative group"

                                :style="{ backgroundColor: cell.subject.color, color: getContrastYIQ(cell.subject.color) }">

                               

                                <div class="font-bold text-xs truncate mb-1">{{ cell.subject.name }}</div>

                                <div class="text-xs opacity-90">{{ cell.cls.type }} | {{ cell.cls.room }}</div>

                               

                                <button @click="removeClass(cell.subject.id, cell.cls.id)"

                                        class="absolute top-1 right-1 bg-red-600 text-white rounded-full w-5 h-5 flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity"

                                        title="Odstranit akci">

                                    &times;

                                </button>

                            </td>

                        </template>

                    </tr>

                </tbody>

<script>

    const { createApp, ref, computed } = Vue;

    createApp({

        setup() {

            const timeBlocks = [

                "7:15-8:00", "8:00-8:45", "9:00-9:45", "9:45-10:30",

                "10:45-11:30", "11:30-12:15", "12:30-13:15", "13:15-14:00",

                "14:15-15:00", "15:00-15:45", "16:00-16:45", "16:45-17:30",

                "17:45-18:30", "18:30-19:15"

            ];

           

            const days = ["pondělí", "úterý", "středa", "čtvrtek", "pátek"];

            const subjects = ref([

                { id: 1, name: "Výroba a užití elektrické energie", color: "#3b82f6", visible: true, classes: [] },

                { id: 2, name: "Jazyk anglický III pro FEI", color: "#ef4444", visible: true, classes: [] },

                { id: 3, name: "Soft Skills I", color: "#10b981", visible: true, classes: [] },

                { id: 4, name: "Fyzika I", color: "#f59e0b", visible: true, classes: [] },

                { id: 5, name: "Matematická analýza 2", color: "#8b5cf6", visible: true, classes: [] },

                { id: 6, name: "Vybrané kap. z el. strojů a přístrojů", color: "#ec4899", visible: true, classes: [] },

                { id: 7, name: "Technická dokumentace", color: "#14b8a6", visible: true, classes: [] },

                { id: 8, name: "Technické vybavení budov", color: "#f97316", visible: true, classes: [] }

            ]);

            const newForm = ref({

                subjectId: "",

                type: "Cvičení",

                room: "",

                day: "pondělí",

                start: 0,

                end: 1

            });

            const errorMsg = ref("");

            // Výpočet mřížky na základě aktivních předmětů

            const computedGrid = computed(() => {

                let grid = {};

                days.forEach(d => grid[d] = new Array(timeBlocks.length).fill(null));

                // Vybereme jen viditelné předměty

                subjects.value.filter(s => s.visible).forEach(subject => {

                    subject.classes.forEach(cls => {

                        let span = cls.end - cls.start + 1;

                       

                        // Zapsání startovní buňky

                        if (grid[cls.day][cls.start] === null) {

                            grid[cls.day][cls.start] = { subject, cls, span };

                            // Zapsání zástupných tokenů pro sloučení buněk (colspan)

                            for (let i = 1; i < span; i++) {

                                if (grid[cls.day][cls.start + i] === null) {

                                    grid[cls.day][cls.start + i] = 'skip';

                                }

                            }

                        }

                    });

                });

                return grid;

            });

            // Přidání nové akce do rozvrhu (obsahuje kontrolu kolizí a validity dat)

            const addClass = () => {

                errorMsg.value = "";

               

                if (newForm.value.end < newForm.value.start) {

                    errorMsg.value = "Konec bloku nesmí předcházet začátku.";

                    return;

                }

                // Kontrola překryvu

                let overlap = false;

                subjects.value.filter(s => s.visible).forEach(subject => {

                    subject.classes.forEach(cls => {

                        if (cls.day === newForm.value.day) {

                            if (newForm.value.start <= cls.end && newForm.value.end >= cls.start) {

                                overlap = true;

                            }

                        }

                    });

                });

                if (overlap) {

                    errorMsg.value = "Detekována kolize! V tento čas již máte (viditelný) předmět.";

                    return;

                }

                const targetSubject = subjects.value.find(s => s.id === newForm.value.subjectId);

                if (targetSubject) {

                    targetSubject.classes.push({

                        id: Date.now(),

                        type: newForm.value.type,

                        room: newForm.value.room,

                        day: newForm.value.day,

                        start: newForm.value.start,

                        end: newForm.value.end

                    });

                   

                    // Reset drobných hodnot ve formuláři, zachování předmětu pro rychlé vkládání

                    newForm.value.room = "";

                }

            };

            const removeClass = (subjectId, classId) => {

                const subject = subjects.value.find(s => s.id === subjectId);

                if (subject) {

                    subject.classes = subject.classes.filter(c => c.id !== classId);

                }

            };

            // Funkce pro zachování čitelnosti textu (černý vs. bílý) na základě HEX barvy pozadí

            const getContrastYIQ = (hexcolor) => {

                hexcolor = hexcolor.replace("#", "");

                var r = parseInt(hexcolor.substr(0,2),16);

                var g = parseInt(hexcolor.substr(2,2),16);

                var b = parseInt(hexcolor.substr(4,2),16);

                var yiq = ((r*299)+(g*587)+(b*114))/1000;

                return (yiq >= 128) ? 'black' : 'white';

            };

            return {

                timeBlocks, days, subjects, newForm, addClass, removeClass, computedGrid, errorMsg, getContrastYIQ

            };

        }

    }).mount('#app');

</script>
